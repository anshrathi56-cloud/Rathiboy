import java.util.*;
class Main {
    public static void main(String[] args) {
        int i=2,n;
        Scanner s=new Scanner(System.in);
        System.out.println("*Program to print all N Even natural numbers*");
        System.out.print("Enter the value of N:");
        n=s.nextInt();
        
        while(i<=n)
        {
        System.out.println(i);
        i+=2;
        }
    }
}
