import java.util.*;
class Main {
    public static void main(String[] args) {
        int i=1,n,sum=0;
        Scanner s=new Scanner(System.in);
        System.out.println("*Program to print ODD of Even natural numbers*");
        System.out.print("Enter the value of N:");
        n=s.nextInt();
        while(i<=n)
        {
        sum=sum+i;
        i+=2;
        }
        System.out.println(sum);
    }
}
