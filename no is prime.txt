import java.util.*;
class Main 
{
    public static void main(String[] args) 
    {
        int n,i,found=1;
        Scanner s=new Scanner(System.in);
        System.out.println("enter the no.");
        n=s.nextInt();
        for(i=2;i<=n-1;i++)
        {
            if(n%i==0)
            {
                found=0;
                break;
            }
        }
        if(found==0)
            System.out.println("not prime");
        else
            System.out.println("prime");
    }
}
